Arquitectura U-Net¶

A rede U-Net é unha arquitectura de rede neuronal convolucional deseñada para tarefas de segmentación de imaxes, inicialmente desenvolvida para segmentación biomédica. A súa estrutura ten forma de "U", estando composta por unha parte de contracción (encoder) e unha parte de expansión (decoder). Debaixo podes ver un esquema da arquitectura orixinal proposta no 2015 por Olaf Ronneberger et al. no artigo "U-Net: Convolutional Networks for Biomedical Image Segmentation".

A parte do codificador segue unha estrutura convencional de rede neuronal convolucional (CNN) para extraer características relevantes das imaxes de entrada. A imaxe de entrada é procesada en bloques, onde se lle aplican dúas capas convolucionais seguidas dunha capa de pooling para reducir a dimensión espacial das características extraídas. Cada capa convolucional utiliza un tamaño de filtro de 3x3 e unha función de activación ReLU para introducir non linealidade no modelo. A capa de pooling emprega unha operación de max-pooling con un tamaño de 2x2 para reducir a resolución das características, o que axuda a diminuír o número de parámetros e a evitar o sobreaxuste. Ao igual que nunha CNN básica, o número de filtros nas capas convolucionais duplícase por dous despois de cada capa de pooling, permitindo á rede capturar características máis complexas a medida que avanza.

A parte do decodificador segue unha estrutura simétrica á do codificador. As características obtidas do codificador son procesadas en bloques, onde se lle aplican un upsampling para aumentar a resolución seguido de dúas capas convolucionais. Cada capa convolucional tamén utiliza un tamaño de filtro de 3x3 e unha función de activación ReLU. Dependendo da implementación, as capa de upsampling consisten nunha operación de reescalado sinxela (utilizando interpolación) ou unha convolución trasposta. Neste caso, o número de filtros nas capas convolucionais do decodificador redúcese á metade despois de cada operación de upsampling, restaurando gradualmente a resolución orixinal da imaxe de entrada.

É tamén relevante que en cada nivel de profundidade da rede, as características do codificador son concatenadas coas características correspondentes do decodificador. Esta técnica, coñecida como "skip connections" ou conexións de salto, axuda a preservar a información espacial e os detalles finos da imaxe orixinal, facilitando unha mellor reconstrucción da imaxe de saída.

class UNet(nn.Module):
    def __init__(self, in_channels=3, out_channels=1, features=[64, 128, 256, 512]):
        super(UNet, self).__init__()
        self.encoder = nn.ModuleList()
        self.decoder = nn.ModuleList()
        # Encoder
        for feature in features:
            self.encoder.append(self.conv_block(in_channels, feature))
            in_channels = feature
        # Decoder
        for feature in reversed(features):
            self.decoder.append(
                nn.ConvTranspose2d(feature*2, feature, kernel_size=2, stride=2)
            )
            self.decoder.append(self.conv_block(feature*2, feature))
        self.bottleneck = self.conv_block(features[-1], features[-1]*2)
        self.final_conv = nn.Conv2d(features[0], out_channels, kernel_size=1)

    def conv_block(self, in_channels, out_channels):
        return nn.Sequential(
            nn.Conv2d(in_channels, out_channels, kernel_size=3, padding=1),
            nn.ReLU(inplace=True),
            nn.Conv2d(out_channels, out_channels, kernel_size=3, padding=1),
            nn.ReLU(inplace=True),
        )

    def forward(self, x):
        skip_connections = []
        for layer in self.encoder:
            x = layer(x)
            skip_connections.append(x)
            x = nn.MaxPool2d(kernel_size=2)(x)
        x = self.bottleneck(x)
        skip_connections = skip_connections[::-1]
        for idx in range(0, len(self.decoder), 2):
            x = self.decoder[idx](x)
            skip_connection = skip_connections[idx//2]
            if x.shape != skip_connection.shape:
                x = TF.resize(x, size=skip_connection.shape[2:])
            x = torch.cat((skip_connection, x), dim=1)
            x = self.decoder[idx+1](x)
        return self.final_conv(x)

from torch.utils.data import Dataset, DataLoader
from PIL import Image #cargar as imaxes
import os

import torchvision.transforms as T

class SegmentationDataset(Dataset):
    def __init__(self, img_dir, masks_dir, size=128):
        self.imagenes_dir = img_dir
        self.masks_dir = masks_dir
        self.imagenes = sorted([f for f in os.listdir(self.imagenes_dir) if f.lower().endswith(('.jpg', '.png', '.jpeg'))])
        self.masks = sorted([f for f in os.listdir(self.masks_dir) if f.lower().endswith(('.jpg', '.png', '.jpeg'))])
        self.transform_img = T.Compose([
            T.Resize((size, size)),
            T.ToTensor(),
        ])
        self.transform_mask = T.Compose([
            T.Resize((size, size), interpolation=Image.NEAREST),
            T.ToTensor(),
        ])

    def __len__(self): # numero de imaxes no conxunto de datos
        return len(self.imagenes)

    def __getitem__(self, idx): # Imaxe e máscara a apartir do índice
        ruta_img = os.path.join(self.imagenes_dir, self.imagenes[idx])
        ruta_mask = os.path.join(self.masks_dir, self.masks[idx])
        imagen = Image.open(ruta_img).convert('RGB')
        mask = Image.open(ruta_mask).convert('L')
        imagen = self.transform_img(imagen)
        mask = self.transform_mask(mask)
        mask = (mask > 0.5).float() # binarizar máscara
        return imagen, mask

import torch
import torch.nn as nn
import numpy as np
from torchvision.models.segmentation import fcn_resnet50, deeplabv3_resnet50
from tqdm import tqdm

# Métricas auxiliares
def dice_coef(pred, target, eps=1e-6):
    pred = (pred > 0.5).float()
    target = (target > 0.5).float()
    inter = (pred * target).sum()
    union = pred.sum() + target.sum()
    return (2 * inter + eps) / (union + eps)

def hausdorff_distance(pred, target):
    # Simple version: max of min distances between edge pixels
    from scipy.spatial.distance import directed_hausdorff
    pred = (pred > 0.5).cpu().numpy().astype(np.uint8)[0]
    target = (target > 0.5).cpu().numpy().astype(np.uint8)[0]
    pred_pts = np.argwhere(pred)
    target_pts = np.argwhere(target)
    if len(pred_pts) == 0 or len(target_pts) == 0:
        return np.nan
    return max(directed_hausdorff(pred_pts, target_pts)[0], directed_hausdorff(target_pts, pred_pts)[0])

# Early stopping
class EarlyStopping:
    def __init__(self, patience=5):
        self.patience = patience
        self.counter = 0
        self.best_loss = np.inf
        self.best_state = None

    def step(self, loss, model):
        if loss < self.best_loss:
            self.best_loss = loss
            self.best_state = {k: v.cpu() for k, v in model.state_dict().items()}
            self.counter = 0
            return False
        else:
            self.counter += 1
            return self.counter >= self.patience

# Adestramento xenérico
def entrenamiento(model, dataloaders, device, num_epochs=30, patience=5):
    criterion = nn.BCELoss()
    optimizer = torch.optim.Adam(model.parameters(), lr=1e-4)
    early_stopping = EarlyStopping(patience)
    model.to(device)
    best_model = None

    for epoch in range(num_epochs):
        model.train() # Adestramento
        train_loss = 0
        for imgs, masks in tqdm(dataloaders['train']):
            imgs, masks = imgs.to(device), masks.to(device)
            optimizer.zero_grad()
            
            if hasattr(model, 'forward'):
                if hasattr(model, 'final_conv'):  # UNet
                    outputs = model(imgs)
                else:  # torchvision models
                    outputs = model(imgs)['out']
            else:
                outputs = model(imgs)['out']
            outputs = outputs.squeeze(1)
            loss = criterion(outputs, masks)
            loss.backward()
            optimizer.step()
            train_loss += loss.item() * imgs.size(0)
        train_loss /= len(dataloaders['train'].dataset)

        # Validación
        model.eval()
        val_loss = 0
        with torch.no_grad():
            for imgs, masks in dataloaders['val']:
                imgs, masks = imgs.to(device), masks.to(device)
                if hasattr(model, 'final_conv'):
                    outputs = model(imgs)
                else:
                    outputs = model(imgs)['out']
                outputs = outputs.squeeze(1)
                loss = criterion(outputs, masks)
                val_loss += loss.item() * imgs.size(0)
        val_loss /= len(dataloaders['val'].dataset)

        print(f"Epoch {epoch+1}: train_loss={train_loss:.4f}, val_loss={val_loss:.4f}")

        if early_stopping.step(val_loss, model):
            print("Early stopping!")
            break

    # Restaurar mellor modelo
    if early_stopping.best_state is not None:
        model.load_state_dict(early_stopping.best_state)
    return model

# Avaliación e visualización
def evaluate_model(model, dataloader, device):
    dices, hausdorffs = [], []
    model.eval()
    with torch.no_grad():
        for imgs, masks in dataloader:
            imgs, masks = imgs.to(device), masks.to(device)
            if hasattr(model, 'final_conv'):
                outputs = model(imgs)
            else:
                outputs = model(imgs)['out']
            outputs = torch.sigmoid(outputs.squeeze(1))
            for i in range(imgs.size(0)):
                dice = dice_coef(outputs[i], masks[i])
                hd = hausdorff_distance(outputs[i].cpu(), masks[i].cpu())
                dices.append(dice.item())
                hausdorffs.append(hd)
    return np.mean(dices), np.nanmean(hausdorffs)

# Visualización TP, TN, FP, FN
def plot_tp_tn_fp_fn(img, mask, pred):
    import matplotlib.pyplot as plt
    img = img.permute(1,2,0).cpu().numpy()
    mask = mask.squeeze().cpu().numpy()
    pred = (pred.squeeze().cpu().numpy() > 0.5).astype(np.uint8)
    tp = (mask == 1) & (pred == 1)
    tn = (mask == 0) & (pred == 0)
    fp = (mask == 0) & (pred == 1)
    fn = (mask == 1) & (pred == 0)
    vis = np.zeros((*mask.shape, 3))
    vis[tp] = [0,1,0]   # TP: green
    vis[fp] = [1,0,0]   # FP: red
    vis[fn] = [0,0,1]   # FN: blue
    vis[tn] = [0,0,0]   # TN: black
    plt.figure(figsize=(12,4))
    plt.subplot(1,3,1); plt.imshow(img); plt.title("Image")
    plt.subplot(1,3,2); plt.imshow(mask, cmap='gray'); plt.title("Mask")
    plt.subplot(1,3,3); plt.imshow(vis); plt.title("TP/FP/FN/TN")
    plt.show()
